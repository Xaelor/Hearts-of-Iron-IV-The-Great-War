
on_actions = {

	on_startup = {
		effect = {
			# Bosnian Politics Soft Fix
			if = {
				limit = {
					has_start_date < 1911.1.1
				}
				BOS = {
					set_politics = {
						ruling_party = neutrality
						last_election = "1910.1.1"
						election_frequency = 48
						elections_allowed = no
					}

					set_popularities = {
						democratic = 15
						fascism = 0
						communism = 0
						neutrality = 85
					}
				}
			}
		}
	}

	on_release_as_free = {
		effect = {
			# Marijan Varešanin Transfer to Croatia
			if = {
				limit = {
					country_exists = CRO
					NOT = {
						CRO = {
							has_character = BOS_marijan_varesanin
						}
					}
				}
				BOS_marijan_varesanin = {
					set_nationality = CRO
				}
			}
		}
	}

	on_release_as_puppet = {
		effect = {
			# Marijan Varešanin Transfer to Croatia
			if = {
				limit = {
					country_exists = CRO
					NOT = {
						CRO = {
							has_character = BOS_marijan_varesanin
						}
					}
				}
				BOS_marijan_varesanin = {
					set_nationality = CRO
				}
			}
		}
	}

}
