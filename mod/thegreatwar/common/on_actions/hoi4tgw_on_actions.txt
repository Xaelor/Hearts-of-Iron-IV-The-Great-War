			if = {
				limit = {

				}
				BOS = {
					set_politics = {
						ruling_party = neutrality
						last_election = "1910.1.1"

		}
	}

				}
			}
		}
	}

	on_war_relation_added = {

							is_in_faction_with = TUR
						}
