	on_startup = {
		effect = {
			# Bosnian Politics Soft Fix
			if = {
				limit = {
					has_start_date < 1911.1.1
				}
				BOS = {
					set_politics = {
						ruling_party = neutrality
						last_election = "1910.1.1"
						election_frequency = 48

					}

					}
				}
			}
		}
	}

	on_release_as_free = {
		effect = {
			# Marijan Varešanin Transfer to Croatia
			if = {
				limit = {
					country_exists = CRO
					NOT = {
						CRO = {
							has_character = BOS_marijan_varesanin
						}
					}
				}
				BOS_marijan_varesanin = {
					set_nationality = CRO
				}
			}
		}
	}

	on_release_as_puppet = {

					set_nationality = CRO
				}
			}
		}
	}

	on_war_relation_added = {
		effect = {
			if = {
				limit = {
					original_tag = ITA

							controls_state = 826
							controls_state = 273

							controls_state = 663
						}

							has_completed_focus = ITA_ww1_italian_askari
						}
						has_war_with_major = yes
						NOT = {
							is_in_faction_with = TUR
						}
					}
				}
				country_event = { id = ww1_libya.1 random_days = 30 }
			}
		}
	}

	on_capitulation_immediate = {

					}

					}
				}
				FROM = {
					annex_country = {
						target = CHI
						transfer_troops = no
					}
				}

				news_event = { id = kis_news.123 hours = 6 }
			}
			if = {
				limit = {
					NOT = {
						has_global_flag = kis_xinhai_revolution_ended
					}
