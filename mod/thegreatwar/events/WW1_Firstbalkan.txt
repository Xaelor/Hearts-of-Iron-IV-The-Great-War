#____________________________________________________________
# Invitation to join the Balkan League sent by FROM
#____________________________________________________________
country_event = {
	id = ww1_firstbalkan.3
	title = ww1_firstbalkan.3.t
	desc = ww1_firstbalkan.3.d
	picture = ww1_firstbalkan_3

	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			FROM = { save_event_target_as = alliance_inviter }
			save_event_target_as = alliance_applicant
		}
	}

	# PREV Yes
	option = {
		name = ww1_firstbalkan.3.a
		ai_chance = {
			base = 100
		}
		if = {
			limit = {
				NOT = {
					original_tag = ROM
					original_tag = ALB
				}
			}
			custom_effect_tooltip = historical_tt
		}
		FROM = {
			faction_leader = {
				add_to_faction = ROOT
			}
			country_event = { id = generic.6 hours = 6 }
		}
		add_ai_strategy = {
			type = alliance
			id = FROM
			value = 200
		}
		hidden_effect = {
			add_opinion_modifier = {
				target = FROM
				modifier = HOI4TGW_Opinion_improved_diplomatic_relations
			}
			reverse_add_opinion_modifier = {
				target = FROM
				modifier = HOI4TGW_Opinion_improved_diplomatic_relations
			}
		}
	}

	# PREV No
	option = {
		name = ww1_firstbalkan.3.b
		ai_chance = {
			base = 0
		}
		FROM = {
			country_event = { id = generic.7 hours = 6 }
		}
		if = {
			limit = {
				original_tag = ROM
				original_tag = ALB
			}
			custom_effect_tooltip = historical_tt
		}
	}
}

# Balkan League Leader receive Peace Offer from Ottoman
#____________________________________________________________
country_event = {
	id = ww1_firstbalkan.6
	title = ww1_firstbalkan.6.t
	desc = ww1_firstbalkan.6.d
	picture = ww1_firstbalkan_6

	fire_only_once = yes
	is_triggered_only = yes

	# BLL Yes
	option = {
		name = ww1_firstbalkan.6.a
		ai_chance = {
			base = 95
		}
		custom_effect_tooltip = historical_tt
		custom_effect_tooltip = generic_skip_one_line_tt
		every_country = {
			limit = {
				has_idea = BLL_liberation_of_the_balkans_ns
			}
			remove_ideas = BLL_liberation_of_the_balkans_ns
		}
		custom_effect_tooltip = generic_skip_one_line_tt
		# If Greece did not participate in the First Balkan War, then region 758 (Central Epirus) will not go to it, but to Albania
		if = {
			limit = {
				NOT = {
					GRE = {
						has_war_with = TUR
					}
				}
			}	
			ALB = {
				transfer_state = 758
				add_state_core = 758
			}		
		}	
		# If Greece did not participate in the First Balkan War, then region 731 (West Macedonia) will not go to it, but to Bulgaria
		# However, if region 731 (West Macedonia) already belongs to Greece (it was transferred by Turkey before the start of the war), then it remains with Greece		
		if = {
			limit = {
				NOT = {
					GRE = {
						has_war_with = TUR
					}
					GRE = {
						owns_state = 731 
					}
				}
			}	
			BUL = {
				transfer_state = 731
				add_state_core = 731
			}			
		}			
		# If Greece participate in the First Balkan War, then region 731 (West Macedonia), 758 (Central Epirus) and 790 (North Aegean islands) will go to it
		if = {
			limit = {
				GRE = {
					has_war_with = TUR
				}
			}	
			GRE = {
				white_peace = TUR
				transfer_state = 731
				transfer_state = 758
				transfer_state = 790
				add_state_core = 731
				add_state_core = 758
				add_state_core = 790
				diplomatic_relation = {
					country = TUR
					relation = non_aggression_pact
				}
			}	
		}	
		# If Bulgaria did not participate in the First Balkan War, then regions 184 (East Macedonia) and 759 (Nordeastern Macedonia) will not go to it, but to Greece and Serbia
		if = {
			limit = {
				NOT = {
					BUL = {
						has_war_with = TUR
					}
				}	
			}		
			GRE = {
				transfer_state = 184
				add_state_core = 184
			}	
			SER = {
				transfer_state = 759
				add_state_core = 759
			}			
		}	
		# If Bulgaria participate in the First Balkan War, then regions 788 (East Rumelia) and 759 (Nordeastern Macedonia) will go to it
		if = {
			limit = {
				BUL = {
					has_war_with = TUR
				}
			}		
			BUL = {
				white_peace = TUR
				transfer_state = 788
				transfer_state = 759
				add_state_core = 788
				add_state_core = 759
				diplomatic_relation = {
					country = TUR
					relation = non_aggression_pact
				}
			}	
		}	
		# If Turkey returned Greece to its national states (184, 756, 757) before the start of the war, then it remains with Greece and not transferred to Bulgaria
		if = {
			limit = {
				TUR = {
					owns_state = 184 
				}	
				TUR = {
					owns_state = 756
				}
				TUR = {
					owns_state = 757
				}	
			}		
			BUL = {
				transfer_state = 184
				transfer_state = 756
				transfer_state = 757
				add_state_core = 184
				add_state_core = 756
				add_state_core = 757
			}	
		}	

		MNT = {
			white_peace = TUR
		}		
		MNT = {
			transfer_state = 761
			add_state_core = 761
			diplomatic_relation = {
				country = TUR
				relation = non_aggression_pact
			}
		}
		ALB = {
			transfer_state = 44
			transfer_state = 803
			add_state_core = 44
			add_state_core = 803
			diplomatic_relation = {
				country = TUR
				relation = non_aggression_pact
			}
		}
		SER = {
			white_peace = TUR
		}		
		SER = {
			transfer_state = 106
			transfer_state = 760
			add_state_core = 106
			add_state_core = 760
			diplomatic_relation = {
				country = TUR
				relation = non_aggression_pact
			}
			remove_from_faction = ITA
		}		
		hidden_effect = {
			set_global_flag = kis_firstbalkanwar_victory
			news_event = {
				hours = 6
				id = ww1_firstbalkan.101
			}
			# Diplomatics relations changed to avoid AI to do stupid stuff
			GRE = {
				diplomatic_relation = {
					country = TUR
					relation = non_aggression_pact
				}
				add_opinion_modifier = {
					target = TUR
					modifier = HOI4TGW_Opinion_hostile_relations_post_balkan_war
				}
			}
			BUL = {
				diplomatic_relation = {
					country = TUR
					relation = non_aggression_pact
				}
				add_opinion_modifier = {
					target = TUR
					modifier = HOI4TGW_Opinion_hostile_relations_post_balkan_war
				}
			}
			MNT = {
				diplomatic_relation = {
					country = TUR
					relation = non_aggression_pact
				}
				add_opinion_modifier = {
					target = TUR
					modifier = HOI4TGW_Opinion_hostile_relations_post_balkan_war
				}
			}
			ALB = {
				diplomatic_relation = {
					country = TUR
					relation = non_aggression_pact
				}
			}
			SER = {
				diplomatic_relation = {
					country = TUR
					relation = non_aggression_pact
				}
				add_opinion_modifier = {
					target = TUR
					modifier = HOI4TGW_Opinion_hostile_relations_post_balkan_war
				}
			}
			TUR = {
				add_opinion_modifier = {
					target = SER
					modifier = HOI4TGW_Opinion_hostile_relations_post_balkan_war
				}
				add_opinion_modifier = {
					target = BUL
					modifier = HOI4TGW_Opinion_hostile_relations_post_balkan_war
				}
				add_opinion_modifier = {
					target = GRE
					modifier = HOI4TGW_Opinion_hostile_relations_post_balkan_war
				}
				add_opinion_modifier = {
					target = MNT
					modifier = HOI4TGW_Opinion_hostile_relations_post_balkan_war
				}
			}
		}
	}

	# BLL No
	option = {
		name = ww1_firstbalkan.6.b
		ai_chance = {
			base = 5
			modifier = {
				factor = 0
				is_historical_focus_on = yes
			}
		}
		custom_effect_tooltip = no_scripted_peace_deal_tt
		every_country = {
			limit = {
				has_idea = BLL_liberation_of_the_balkans_ns
			}
			remove_ideas = BLL_liberation_of_the_balkans_ns
		}
	}
}
